REST API Endpoints

POST /api/optimize — Single-engine optimization 
{ "objective": "(p) => (p.x)**2", "parameters": [{"name":"x","min":-10,"max":10}], "maxEvals": 1000 }
POST /api/dual-optimize — Dual-engine with cross-pollination 
{ "objective": "...", "parameters": [...], "maxEvals": 1000, "cycles": 5 }
GET /api/status/:jobId — Check job status
GET /api/result/:jobId — Get completed results
GET /api/health — Server health check
GET /api/billing/tiers — Pricing tiers

SDK Usage

const { optimize, dualOptimize, exportResult } = require('aegis-optimizer');

const result = await optimize({
  objective: (p) => yourCostFunction(p.x, p.y),
  parameters: [
    { name: 'x', min: 0, max: 100 },
    { name: 'y', min: -50, max: 50, type: 'integer' },
    { name: 'mode', type: 'categorical', values: ['fast','balanced','thorough'] },
  ],
  constraints: [
    (p) => Math.max(0, p.x + p.y - 100),  // x + y ≤ 100
    { type: 'range', param: 'x', min: 0, max: 50 },
  ],
  maxEvals: 5000,
  warmStart: previousResult.best,  // optional
});

console.log(result.best);
console.log(exportResult(result, 'summary'));
